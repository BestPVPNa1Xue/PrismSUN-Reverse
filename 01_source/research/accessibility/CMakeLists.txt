cmake_minimum_required(VERSION 3.16)
project(accessibility_reconstruction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(accessibility_reconstruction SHARED src/accessibility_core.cpp)
target_include_directories(accessibility_reconstruction PUBLIC src)
target_compile_options(accessibility_reconstruction PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    target_sources(accessibility_reconstruction PRIVATE src/accessibility_jni.cpp src/accessibility_env.cpp src/accessibility_monitor.cpp)
else()
    add_executable(accessibility_demo src/accessibility_demo.cpp src/accessibility_core.cpp)
    target_compile_options(accessibility_demo PRIVATE -Wall -Wextra -Werror)
endif()
