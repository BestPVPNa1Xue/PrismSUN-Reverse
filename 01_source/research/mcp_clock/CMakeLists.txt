cmake_minimum_required(VERSION 3.16)
project(prism_mcp_clock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(prism_mcp_chain STATIC src/mcp_chain.cpp)
add_library(prism_clock_chain STATIC src/clock_chain.cpp src/crypto.cpp)
target_include_directories(prism_mcp_chain PUBLIC src)
target_include_directories(prism_clock_chain PUBLIC src)
if(WIN32)
  target_link_libraries(prism_clock_chain PUBLIC bcrypt)
endif()
add_executable(prism_mcp_clock src/main.cpp)
target_link_libraries(prism_mcp_clock PRIVATE prism_mcp_chain prism_clock_chain)
add_executable(unit_tests tests/unit_tests.cpp)
target_link_libraries(unit_tests PRIVATE prism_mcp_chain prism_clock_chain)
enable_testing()
add_test(NAME chains COMMAND unit_tests)
