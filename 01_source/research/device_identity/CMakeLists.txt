cmake_minimum_required(VERSION 3.16)
project(prism_identity_reconstruction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(prism_identity SHARED src/crypto.cpp src/identity.cpp src/android_adapter.cpp)
target_include_directories(prism_identity PUBLIC src)
target_compile_options(prism_identity PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  target_link_libraries(prism_identity PRIVATE dl)
endif()
add_executable(prism_identity_demo src/main.cpp src/crypto.cpp src/identity.cpp)
target_compile_options(prism_identity_demo PRIVATE -Wall -Wextra -Werror)
