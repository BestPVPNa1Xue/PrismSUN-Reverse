cmake_minimum_required(VERSION 3.16)
project(prism_environment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(prism_environment_core STATIC src/environment_memory.cpp ../src/observed_models.cpp ../src/class_probe_model.cpp)
target_include_directories(prism_environment_core PUBLIC src ../src)
add_executable(prism_environment src/main.cpp)
target_link_libraries(prism_environment PRIVATE prism_environment_core)
if(ANDROID)
  set_target_properties(prism_environment_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(prism_environment_jni SHARED src/input_methods_jni.cpp ../accessibility/src/accessibility_env.cpp)
  target_link_libraries(prism_environment_jni PRIVATE prism_environment_core)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(environment_tests tests/unit_tests.cpp)
  target_link_libraries(environment_tests PRIVATE prism_environment_core)
  add_test(NAME environment_models COMMAND environment_tests)
endif()
